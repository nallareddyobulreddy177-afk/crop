-- 1. PROFILES (Farmers)
CREATE TABLE public.profiles (
  id UUID REFERENCES auth.users ON DELETE CASCADE PRIMARY KEY,
  full_name TEXT NOT NULL,
  phone_number TEXT UNIQUE,
  preferred_language TEXT DEFAULT 'te' CHECK (preferred_language IN ('en', 'te', 'hi')),
  created_at TIMESTAMPTZ DEFAULT NOW()
);

-- 2. FIELDS (Farm Plots & PostGIS Geometries)
CREATE TABLE public.fields (
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  user_id UUID REFERENCES public.profiles(id) ON DELETE CASCADE NOT NULL,
  field_name TEXT NOT NULL,
  crop_type TEXT NOT NULL,
  area_acres NUMERIC(5,2) CHECK (area_acres > 0),
  location GEOGRAPHY(POINT, 4326),
  health_score INT DEFAULT 100 CHECK (health_score BETWEEN 0 AND 100),
  created_at TIMESTAMPTZ DEFAULT NOW()
);

-- 3. CROP ANALYSES (Multi-modal Telemetry)

  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,

  image_url TEXT NOT NULL,
  voice_transcript TEXT,
  detected_crop TEXT NOT NULL,
  possible_disease TEXT NOT NULL,
  confidence_score NUMERIC(5,2) NOT NULL,
  severity TEXT CHECK (severity IN ('Low', 'Moderate', 'Severe', 'Critical')),
  observed_symptoms JSONB NOT NULL DEFAULT '[]'::jsonb,
  recommended_actions JSONB NOT NULL DEFAULT '[]'::jsonb,
  preventive_measures JSONB NOT NULL DEFAULT '[]'::jsonb,
  is_expert_escalated BOOLEAN DEFAULT FALSE,

Q1: What dataset did you use to train or fine-tune your vision model? ▾
We utilized PlantVillage combined with regional agricultural extension field samples (~54,000 annotated images). For complex ambiguous cases, we route inputs through multi-modal prompts combining visual descriptors with ambient weather context via Gemini 3.1 Vision.
Q2: How does the app perform in zero-connectivity rural zones? ▾
AgriPulse utilizes an offline-first Service Worker with IndexedDB caching. Leaf scans and voice inputs are saved locally with GPS coordinates and automatically sync to Supabase via background triggers as soon as 2G/3G signal is restored.
Q3: How do you prevent dangerous pesticide recommendations? ▾
We strictly enforce responsible agricultural guidance. The system never generates chemical dosages or unverified formulas. Instead, it defaults to cultural practices, crop sanitation, airflow management, and official extension office contact numbers.
Q4: Why build on Supabase instead of traditional custom backends? ▾
Supabase gives us instant PostgreSQL with PostGIS support for regional hotspot maps, built-in Row Level Security for farmer data privacy, Supabase Storage for leaf images, and Realtime subscriptions for disease alert broadcasts with minimal infrastructure overhead.
